unit janTipHtml; // new laatste voor test met html editor

{$MODE Delphi}

interface

uses
  Windows, LCLIntf, LCLType, LMessages, Messages, SysUtils, Classes, Graphics, Controls, Forms, Dialogs,
  ExtCtrls, ImgList, StdCtrls, Buttons, ColorBox;

type
  TBalloonTipPosition = (btpNone, btpTopLeft, btpTopRight, btpBottomLeft, btpBottomRight, btpTopCenter, btpBottomCenter, btpLeftCenter, btpRightCenter);
  TTextAlignment = (taLeft, taCenter, taRight);
  TIconClickEvent = procedure(Sender: TObject) of object;

  TjanTipHtml = class(TCustomControl)
  private
    FStrings: TStringList;
    FTextAlignment: TTextAlignment;
    FTextColor: TColor;
    FDefaultFontSize: Integer;
    FDefaultFontColor: TColor;
    FEditor: TMemo;
    FToolbar: TPanel;
    procedure SetStrings(const Value: TStringList);
    function GetStrings: TStringList;
    procedure SetDefaults(AColor: TColor; AFontSize: Integer);
    function ColorFromNameOrHex(const AValue: string): TColor;
    function IsValidTag(const Line, Tag: string): Boolean;
    procedure ApplyFormatting(const Tag: string);
    procedure CreateToolbar;
    procedure ApplyBold(Sender: TObject);
    procedure ApplyItalic(Sender: TObject);
    procedure ApplyUnderline(Sender: TObject);
    procedure ApplyColor(Sender: TObject);
  public
    constructor Create(AOwner: TComponent); override;
    destructor Destroy; override;
    procedure ParseAndDrawHTML(Canvas: TCanvas; const ARect: TRect);
    procedure ShowEditor;
    procedure HideEditor;
    procedure SaveAsHtml;
    property Strings: TStringList read GetStrings write SetStrings;
    property TextAlignment: TTextAlignment read FTextAlignment write FTextAlignment;
    property TextColor: TColor read FTextColor write FTextColor;
    property DefaultFontSize: Integer read FDefaultFontSize write FDefaultFontSize;
    property DefaultFontColor: TColor read FDefaultFontColor write FDefaultFontColor;
  end;

implementation

constructor TjanTipHtml.Create(AOwner: TComponent);
begin
  inherited Create(AOwner);
  FStrings := TStringList.Create;
  FTextAlignment := taLeft;
  FTextColor := clBlack;
  FDefaultFontSize := 10;
  FDefaultFontColor := clBlack;
  FEditor := nil;
  FToolbar := nil;
end;

destructor TjanTipHtml.Destroy;
begin
  FStrings.Free;
  if Assigned(FEditor) then
    FEditor.Free;
  if Assigned(FToolbar) then
    FToolbar.Free;
  inherited Destroy;
end;

procedure TjanTipHtml.SetDefaults(AColor: TColor; AFontSize: Integer);
begin
  FDefaultFontColor := AColor;
  FDefaultFontSize := AFontSize;
end;

function TjanTipHtml.ColorFromNameOrHex(const AValue: string): TColor;
begin
  if SameText(AValue, 'clRed') then
    Result := clRed
  else if SameText(AValue, 'clBlack') then
    Result := clBlack
  else if SameText(AValue, 'clNavy') then
    Result := clNavy
  else if SameText(AValue, 'clGreen') then
    Result := clGreen
  else if SameText(AValue, 'clYellow') then
    Result := clYellow
  else if SameText(AValue, 'clBlue') then
    Result := clBlue
  else if SameText(AValue, 'clWhite') then
    Result := clWhite
  else if (Length(AValue) = 6) and TryStrToInt('$' + AValue, Integer(Result)) then
    Result := StringToColor('$00' + AValue) // Hex-waarde
  else
    Result := clBlack; // Fallback naar zwart
end;

function TjanTipHtml.IsValidTag(const Line, Tag: string): Boolean;
var
  OpenTag, CloseTag: string;
begin
  OpenTag := '<' + Tag + '>';
  CloseTag := '</' + Tag + '>';
  Result := (Pos(OpenTag, Line) > 0) and (Pos(CloseTag, Line) > 0);
end;

procedure TjanTipHtml.ApplyFormatting(const Tag: string);
begin
  if Assigned(FEditor) and (FEditor.SelLength > 0) then
    FEditor.SelText := '<' + Tag + '>' + FEditor.SelText + '</' + Tag + '>';
end;

procedure TjanTipHtml.ApplyColor(Sender: TObject);
var
  ColorBox: TColorBox;
begin
  if (Sender is TColorBox) and Assigned(FEditor) and (FEditor.SelLength > 0) then
  begin
    ColorBox := TColorBox(Sender);
    FEditor.SelText := Format('<color=%s>%s</color>', [ColorToString(ColorBox.Selected), FEditor.SelText]);
  end;
end;

procedure TjanTipHtml.CreateToolbar;
var
  BoldButton, ItalicButton, UnderlineButton: TSpeedButton;
  ColorBox: TColorBox;
begin
  if Assigned(FToolbar) then
    Exit;

  FToolbar := TPanel.Create(Self);
  FToolbar.Parent := Self;
  FToolbar.Align := alTop;
  FToolbar.Height := 30;

  BoldButton := TSpeedButton.Create(FToolbar);
  BoldButton.Parent := FToolbar;
  BoldButton.Caption := 'B';
  BoldButton.Font.Style := [fsBold];
  BoldButton.Left := 5;
  BoldButton.Top := 5;
  BoldButton.Width := 30;
  BoldButton.OnClick := @ApplyBold;

  ItalicButton := TSpeedButton.Create(FToolbar);
  ItalicButton.Parent := FToolbar;
  ItalicButton.Caption := 'I';
  ItalicButton.Font.Style := [fsItalic];
  ItalicButton.Left := 40;
  ItalicButton.Top := 5;
  ItalicButton.Width := 30;
  ItalicButton.OnClick := @ApplyItalic;

  UnderlineButton := TSpeedButton.Create(FToolbar);
  UnderlineButton.Parent := FToolbar;
  UnderlineButton.Caption := 'U';
  UnderlineButton.Font.Style := [fsUnderline];
  UnderlineButton.Left := 75;
  UnderlineButton.Top := 5;
  UnderlineButton.Width := 30;
  UnderlineButton.OnClick := @ApplyUnderline;

  ColorBox := TColorBox.Create(FToolbar);
  ColorBox.Parent := FToolbar;
  ColorBox.Left := 110;
  ColorBox.Top := 5;
  ColorBox.Width := 100;
  ColorBox.OnChange := @ApplyColor;
end;

procedure TjanTipHtml.ApplyBold(Sender: TObject);
begin
  ApplyFormatting('b');
end;

procedure TjanTipHtml.ApplyItalic(Sender: TObject);
begin
  ApplyFormatting('i');
end;

procedure TjanTipHtml.ApplyUnderline(Sender: TObject);
begin
  ApplyFormatting('u');
end;

function TjanTipHtml.GetStrings: TStringList;
begin
  if Assigned(FEditor) and FEditor.Visible then
    FStrings.Text := FEditor.Text;
  Result := FStrings;
end;

procedure TjanTipHtml.SetStrings(const Value: TStringList);
begin
  FStrings.Assign(Value);
  if Assigned(FEditor) then
    FEditor.Text := FStrings.Text;
  Invalidate; // Zorg dat de inhoud wordt bijgewerkt
end;

procedure TjanTipHtml.ShowEditor;
begin
  if not Assigned(FEditor) then
  begin
    FEditor := TMemo.Create(Self);
    FEditor.Parent := Self;
    FEditor.Align := alClient;
    CreateToolbar;
  end;
  FEditor.Text := FStrings.Text; // Synchroniseer met Strings
  FEditor.Visible := True;
  if Assigned(FToolbar) then
    FToolbar.Visible := True;
end;

procedure TjanTipHtml.HideEditor;
begin
  if Assigned(FEditor) then
  begin
    FStrings.Text := FEditor.Text; // Synchroniseer terug naar Strings
    FEditor.Visible := False;
  end;
  if Assigned(FToolbar) then
    FToolbar.Visible := False;
end;

procedure TjanTipHtml.SaveAsHtml;
begin
  if Assigned(FEditor) then
    FStrings.Text := FEditor.Text; // Sla de aangepaste HTML op
end;

procedure TjanTipHtml.ParseAndDrawHTML(Canvas: TCanvas; const ARect: TRect);
var
  i, X, Y: Integer;
  Line, Segment, ColorValue, SizeStr: string;
  Bold, Italic, Underline: Boolean;
  CurrentColor: TColor;
  FontSize, DefaultFontSize: Integer;
  TextWidth, StartX: Integer;
begin
  Y := ARect.Top + 10;
  CurrentColor := FDefaultFontColor;
  DefaultFontSize := FDefaultFontSize;
  Canvas.Brush.Style := bsClear;
  Canvas.Font.Size := DefaultFontSize;

  for i := 0 to FStrings.Count - 1 do
  begin
    Line := FStrings[i];
    Bold := False;
    Italic := False;
    Underline := False;

    TextWidth := Canvas.TextWidth(StripHTMLTags(Line));
    case FTextAlignment of
      taLeft:
        StartX := ARect.Left;
      taCenter:
        StartX := ARect.Left + (ARect.Width - TextWidth) div 2;
      taRight:
        StartX := ARect.Right - TextWidth;
    end;

    X := StartX;

    while Line <> '' do
    begin
      if Pos('<b>', Line) = 1 then
      begin
        Bold := True;
        Delete(Line, 1, 3);
        Continue;
      end
      else if Pos('</b>', Line) = 1 then
      begin
        Bold := False;
        Delete(Line, 1, 4);
        Continue;
      end
      else if Pos('<i>', Line) = 1 then
      begin
        Italic := True;
        Delete(Line, 1, 3);
        Continue;
      end
      else if Pos('</i>', Line) = 1 then
      begin
        Italic := False;
        Delete(Line, 1, 4);
        Continue;
      end
      else if Pos('<u>', Line) = 1 then
      begin
        Underline := True;
        Delete(Line, 1, 3);
        Continue;
      end
      else if Pos('</u>', Line) = 1 then
      begin
        Underline := False;
        Delete(Line, 1, 4);
        Continue;
      end
      else if Pos('<color=', Line) = 1 then
      begin
        ColorValue := Copy(Line, 8, Pos('>', Line) - 8);
        CurrentColor := ColorFromNameOrHex(ColorValue);
        Delete(Line, 1, Pos('>', Line));
        Continue;
      end
      else if Pos('</color>', Line) = 1 then
      begin
        CurrentColor := FDefaultFontColor;
        Delete(Line, 1, 8);
        Continue;
      end
      else if Pos('<font size="', Line) = 1 then
      begin
        SizeStr := Copy(Line, 13, Pos('"', Line, 13) - 13);
        if TryStrToInt(SizeStr, FontSize) then
          Canvas.Font.Size := FontSize;
        Delete(Line, 1, Pos('>', Line));
        Continue;
      end
      else if Pos('</font>', Line) = 1 then
      begin
        Canvas.Font.Size := DefaultFontSize;
        Delete(Line, 1, 7);
        Continue;
      end
      else if Pos('<br>', Line) = 1 then
      begin
        Inc(Y, Canvas.TextHeight('Wg'));
        X := ARect.Left;
        Delete(Line, 1, 4);
        Continue;
      end;

      Segment := Copy(Line, 1, Pos('<', Line + '<') - 1);
      Delete(Line, 1, Length(Segment));

      Canvas.Font.Style := [];
      if Bold then
        Canvas.Font.Style := Canvas.Font.Style + [fsBold];
      if Italic then
        Canvas.Font.Style := Canvas.Font.Style + [fsItalic];
      if Underline then
        Canvas.Font.Style := Canvas.Font.Style + [fsUnderline];

      Canvas.Font.Color := CurrentColor;
      Canvas.TextOut(X, Y, Segment);
      Inc(X, Canvas.TextWidth(Segment));
    end;

    Inc(Y, Canvas.TextHeight('Wg'));
    X := ARect.Left;
  end;

  Canvas.Font.Style := [];
end;

end.
